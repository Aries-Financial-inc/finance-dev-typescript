# PredMdContract

A single tradable event contract at a specific strike.

## Example Usage

```typescript
import { PredMdContract } from "@ariesdotcom/finance-dev/models";
import { Decimal } from "@ariesdotcom/finance-dev/types";

let value: PredMdContract = {
  baseEventId: "<id>",
  candidateStatus: "C",
  eventId: "<id>",
  frontRunner: false,
  incumbent: true,
  lastSyncedAt: new Date("2026-08-25T20:00:03.982Z"),
  nonTradable: false,
  party: "O",
  photo: {
    credit: "Gage Skidmore",
    license: "CC BY-SA 4.0",
    licenseUrl: "https://creativecommons.org/licenses/by-sa/4.0",
    sourceUrl: "https://commons.wikimedia.org/wiki/File:K_Paxton.jpg",
    url: "https://vain-bonnet.org/",
  },
  state: "Arkansas",
  symbol: "<value>",
  symbolSubTypes: [
    "<value 1>",
    "<value 2>",
  ],
  tickSize: new Decimal("0.01"),
  tradable: false,
};
```
