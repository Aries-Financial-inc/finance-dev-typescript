# PredMdContract

A single tradable event contract at a specific strike.

## Example Usage

```typescript
import { PredMdContract } from "finance-dev/models";
import { Decimal } from "finance-dev/types";

let value: PredMdContract = {
  baseEventId: "<id>",
  eventId: "<id>",
  lastSyncedAt: new Date("2026-04-10T17:48:12.596Z"),
  nonTradable: false,
  state: "Utah",
  symbol: "<value>",
  symbolSubTypes: [
    "<value 1>",
  ],
  tickSize: new Decimal("0.01"),
  tradable: false,
};
```
