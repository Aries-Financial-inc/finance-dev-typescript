# EconomicsCalendarEvent

Economic calendar event with its schedule, reported and forecast values.

## Example Usage

```typescript
import { EconomicsCalendarEvent } from "finance-dev/models";

let value: EconomicsCalendarEvent = {
  calendarID: "te_395222",
  category: "Continuing Jobless Claims",
  country: "United States",
  date: new Date("2026-03-05"),
  event: "Continuing Jobless Claims",
  forecast: "1850K",
  importance: 1,
  lastUpdate: "2026-03-03T10:49:26.58",
  previous: "1833K",
  reference: "Feb/21",
  referenceDate: new Date("2026-02-21T00:00:00Z"),
  sourceUrl: "http://www.dol.gov",
  symbol: "UNITEDSTACONJOBCLA",
  teForecast: "1840.0K",
  ticker: "UNITEDSTACONJOBCLA",
  time: "13:30:00",
  unit: "K",
};
```
