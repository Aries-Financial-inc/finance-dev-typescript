  format: "<value>",

  hasSeconds: false,
  hasTicks: true,
  hasWeeklyAndMonthly: true,
  intradayMultipliers: [
    "<value 1>",
  ],
  listedExchange: "<value>",
  minmov: 6606.35,
  monthlyMultipliers: [],
  name: "<value>",
  pricescale: 151834,
  session: "<value>",
  subsessionId: "<id>",
  subsessions: [
    {
      description: "regarding around down below",
      id: "<id>",
      session: "<value>",
    },
  ],
  supportedResolutions: [
    "<value 1>",
    "<value 2>",
    "<value 3>",
  ],
  ticker: "<value>",

  visiblePlotsSet: "<value>",
  volumePrecision: 3265.39,
  weeklyMultipliers: [
    "<value 1>",

    "<value 3>",
  ],
};
