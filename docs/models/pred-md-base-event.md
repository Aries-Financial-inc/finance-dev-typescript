# PredMdBaseEvent

## Example Usage

```typescript
import { PredMdBaseEvent } from "@ariesdotcom/finance-dev/models";

let value: PredMdBaseEvent = {
  baseEventId: "<id>",
  categoryLevels: [
    "Economic Indicators",
    "Labor",
    "United States",
  ],
  displayName: "US Unemployment Rate",
  firstSeenAt: new Date("2026-12-13T16:35:39.458Z"),
  lastSyncedAt: new Date("2024-04-24T02:13:54.727Z"),
};
```
