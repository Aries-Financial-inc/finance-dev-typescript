# PreviewOrderResponse

Response schema for order preview (PreviewOrdResponse)

## Example Usage

```typescript
import { PreviewOrderResponse } from "@ariesdotcom/finance-dev/models";

let value: PreviewOrderResponse = {
  buyingPowerImpact: "1500.01",
  commission: "0.00",
  estimatedCost: "1500.00",
  estimatedMargin: "0.00",
  fees: "0.01",
  numDayTrades: 0,
  optionFees: "0.70",
  optionPremium: "550.00",
  optionRequirement: "0.00",
  warnRuleId: "0",
};
```
