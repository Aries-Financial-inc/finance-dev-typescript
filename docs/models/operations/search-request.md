# SearchRequest

## Example Usage

```typescript
import { SearchRequest } from "@ariesdotcom/finance-dev/models/operations";

let value: SearchRequest = {
  q: "unemployment june",
  state: "ACTIVE",
  baseEventId: "UNR",
  eventId: "UNR_0822",
  category: "Elections > United States House",
  page: 1,
  perPage: 20,
};
```
